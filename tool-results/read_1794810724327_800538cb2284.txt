   249→   249→  private tapL1(x: number): void {
   250→   250→    const l = this.l1;
   251→   251→    if (l.merged || l.done) return;
   252→   252→    const side = x < W / 2 ? -1 : 1; // -1 left, 1 right
   253→   253→    // NOTE: positive angle = clockwise = RIGHT side down, LEFT side up
   254→   254→    const highSide = l.angle > 0.035 ? -1 : l.angle < -0.035 ? 1 : (Math.random() > 0.5 ? -1 : 1);
   255→   255→    const good = side === highSide || Math.abs(l.angle) <= 0.035;
   256→   256→    if (good) {
   257→   257→      l.count++;
   258→   258→      l.angVel += side * 1.5;
   259→   259→      if (side < 0) l.bounceL = 1; else l.bounceR = 1;
   260→   260→      sfxTap(1 + (l.count % 3) * 0.12);
   261→   261→      sfxBoing(side < 0 ? 1.1 : 0.9);
   262→   262→      sayNumber(Math.min(8, l.count));
   263→   263→      spawnSparkles(this.particles, W / 2 + side * 180, 620, 6, PAL.goldLight);
   264→   264→      if (l.count === 4) say('Keep going! Keep going!', { rate: 1.15 });
   265→   265→      if (l.count >= 8) {
   266→   266→        l.merged = true;
   267→   267→        l.mergeT = 0;
   268→   268→        say('And double it! Four plus four equals eight!', { rate: 1.05 });
   269→   269→        sfxSparkle();
   270→   270→        spawnSparkles(this.particles, W / 2, 560, 50, PAL.goldLight);
   271→   271→      }
   272→   272→    } else {
   273→   273→      l.angVel += side * 1.9;
   274→   274→      sfxBonk();
   275→   275→      spawnSparkles(this.particles, W / 2 + side * 180, 620, 8, PAL.redLight);
   276→   276→    }
   277→   277→  }

   279→   279→  private updateL1(dt: number): void {
   280→   280→    const l = this.l1;
   281→   281→    l.windT += dt;
   282→   282→    if (l.bounceL > 0) l.bounceL = Math.max(0, l.bounceL - dt * 4);
   283→   283→    if (l.bounceR > 0) l.bounceR = Math.max(0, l.bounceR - dt * 4);
   284→   284→    if (!l.merged) {
   285→   285→      // destabilizing beam physics + gentle wind gusts (tuned to be fair)
   286→   286→      l.angVel += (l.angle * 1.2 + Math.sin(l.windT * 1.7) * 0.35) * dt * 2.0;
   287→   287→      l.angVel *= 0.955;
   288→   288→      l.angle += l.angVel * dt * 0.5;
   289→   289→      if (Math.abs(l.angle) > 0.72) {
   290→   290→        this.enterRetry('THE BLOCKS SLID OFF!');
   291→   291→        return;
   292→   292→      }
   293→   293→    } else {
   294→   294→      l.mergeT += dt;
   295→   295→      if (l.mergeT > 2.4 && !l.done) {
   296→   296→        l.done = true;
   297→   297→        this.enterReveal(['4', '+', '4', '=', '8'], () => {
   298→   298→          this.enterIntro(2);
   299→   299→        });
   300→   300→      }
   301→   301→    }
   302→   302→  }

   371→   371→  private updateL2(dt: number): void {
   372→   372→    const l = this.l2;
   373→   373→    l.laneX += (l.targetX - l.laneX) * Math.min(1, dt * 6);
   374→   374→    if (!l.done) {
   375→   375→      l.y -= 42 * dt; // auto climb
   376→   376→      if (l.y < 300) l.y = 300;
   377→   377→      l.climbBob += dt;
   378→   378→      // spawn apples
   379→   379→      l.spawnT -= dt;
   380→   380→      if (l.spawnT <= 0 && l.caught + l.apples.filter((a) => !a.caught && !a.dead).length < 9) {
   381→   381→        const lanes = [168, 288, 408];
   382→   382→        l.apples.push({
   383→   383→          x: lanes[Math.floor(Math.random() * 3)] + (Math.random() * 30 - 15),
   384→   384→          y: -40, vy: 150 + Math.random() * 70, caught: false, dead: false, t: 0,
   385→   385→        });
   386→   386→        l.spawnT = Math.max(0.45, 1.1 - this.globalT * 0.01) + Math.random() * 0.5;
   387→   387→      }
   388→   388→      // move & catch
   389→   389→      for (const a of l.apples) {
   390→   390→        a.t += dt;
   391→   391→        if (a.dead) continue;
   392→   392→        if (a.caught) {
   393→   393→          a.x += (l.laneX - a.x) * dt * 8;
   394→   394→          a.y += (l.y - 150 - a.y) * dt * 8;
   395→   395→          if (a.y < l.y - 145) a.dead = true;
   396→   396→        } else {
   397→   397→          a.y += a.vy * dt;
   398→   398→          const dx = a.x - l.laneX;
   399→   399→          const dy = a.y - (l.y - 110);
   400→   400→          if (dx * dx + dy * dy < 80 * 80) {
   401→   401→            a.caught = true;
   402→   402→            l.caught++;
   403→   403→            sfxPop(1 + l.caught * 0.07);
   404→   404→            sayNumber(l.caught);

  1267→  1267→  private loop(ts: number): void {

  1289→  1289→    updateParticles(this.particles, dt * this.timeScale, 300);
