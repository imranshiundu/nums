   253→   253→  private tapL1(x: number): void {
   254→   254→    const l = this.l1;
   255→   255→    if (l.merged || l.done) return;
   256→   256→    const side = x < W / 2 ? -1 : 1; // -1 left, 1 right
   257→   257→    // NOTE: positive angle = clockwise = RIGHT side down, LEFT side up
   258→   258→    const highSide = l.angle > 0.035 ? -1 : l.angle < -0.035 ? 1 : (Math.random() > 0.5 ? -1 : 1);
   259→   259→    const good = side === highSide || Math.abs(l.angle) <= 0.035;
   260→   260→    if (good) {
   261→   261→      l.count++;
   262→   262→      l.angVel += side * 1.5;
   263→   263→      if (side < 0) l.bounceL = 1; else l.bounceR = 1;
   264→   264→      sfxTap(1 + (l.count % 3) * 0.12);
   265→   265→      sfxBoing(side < 0 ? 1.1 : 0.9);
   266→   266→      sayNumber(Math.min(8, l.count));
   267→   267→      spawnSparkles(this.particles, W / 2 + side * 180, 620, 6, PAL.goldLight);
   268→   268→      if (l.count >= 8) {
   269→   269→        l.merged = true;
   270→   270→        l.mergeT = 0;
   271→   271→        say('And double it! Four plus four equals eight!', { rate: 1.05 });
   272→   272→        sfxSparkle();
   273→   273→        spawnSparkles(this.particles, W / 2, 560, 50, PAL.goldLight);
   274→   274→      }
   275→   275→    } else {
   276→   276→      l.angVel += side * 1.9;
   277→   277→      sfxBonk();
   278→   278→      spawnSparkles(this.particles, W / 2 + side * 180, 620, 8, PAL.redLight);
   279→   279→    }
   280→   280→  }
   281→   281→
   282→   282→  private updateL1(dt: number): void {
   283→   283→    const l = this.l1;
   284→   284→    l.windT += dt;
   285→   285→    if (l.bounceL > 0) l.bounceL = Math.max(0, l.bounceL - dt * 4);
   286→   286→    if (l.bounceR > 0) l.bounceR = Math.max(0, l.bounceR - dt * 4);
   287→   287→    if (!l.merged) {
   288→   288→      // destabilizing beam physics + gentle wind gusts (tuned to be fair)
   289→   289→      l.angVel += (l.angle * 1.2 + Math.sin(l.windT * 1.7) * 0.35) * dt * 2.0;
   290→   290→      l.angVel *= 0.955;
   291→   291→      l.angle += l.angVel * dt * 0.5;
   292→   292→      if (Math.abs(l.angle) > 0.72) {
   293→   293→        this.enterRetry('THE BLOCKS SLID OFF!');
   294→   294→        return;
   295→   295→      }
   296→   296→    } else {
   297→   297→      l.mergeT += dt;
   298→   298→      if (l.mergeT > 2.4 && !l.done) {
   299→   299→        l.done = true;
   300→   300→        this.enterReveal(['4', '+', '4', '=', '8'], () => {
   301→   301→          this.enterIntro(2);
   302→   302→        });
   303→   303→      }
   304→   304→    }
   305→   305→  }

   371→   371→  private updateL2(dt: number): void {
   372→   372→    const l = this.l2;
   373→   373→    l.laneX += (l.targetX - l.laneX) * Math.min(1, dt * 6);
   374→   374→    if (!l.done) {
   375→   375→      l.y -= 42 * dt; // auto climb
   376→   376→      if (l.y < 300) l.y = 300;
   377→   377→      l.climbBob += dt;
   378→   378→      // spawn apples
   379→   379→      l.spawnT -= dt;
   380→   380→      if (l.spawnT <= 0 && l.caught + l.apples.filter((a) => !a.caught && !a.dead).length < 9) {
   381→   381→        const lanes = [168, 288, 408];
   382→   382→        l.apples.push({
   383→   383→          x: lanes[Math.floor(Math.random() * 3)] + (Math.random() * 30 - 15),
   384→   384→          y: -40, vy: 150 + Math.random() * 70, caught: false, dead: false, t: 0,
   385→   385→        });
   386→   386→        l.spawnT = Math.max(0.45, 1.1 - this.globalT * 0.01) + Math.random() * 0.5;
   387→   387→      }
   388→   388→      // move & catch
   389→   389→      for (const a of l.apples) {
   390→   390→        a.t += dt;
   391→   391→        if (a.dead) continue;
   392→   392→        if (a.caught) {
   393→   393→          a.x += (l.laneX - a.x) * dt * 8;
   394→   394→          a.y += (l.y - 150 - a.y) * dt * 8;
   395→   395→          if (a.y < l.y - 145) a.dead = true;
   396→   396→        } else {
   397→   397→          a.y += a.vy * dt;
   398→   398→          const dx = a.x - l.laneX;
   399→   399→          const dy = a.y - (l.y - 110);
   400→   400→          if (dx * dx + dy * dy < 80 * 80) {
   401→   401→            a.caught = true;
   402→   402→            l.caught++;
   403→   403→            sfxPop(1 + l.caught * 0.07);
   404→   404→            sayNumber(l.caught);
   405→   405→            spawnSparkles(this.particles, a.x, a.y, 12, PAL.goldLight);
   406→   406→            if (l.caught === 8) {
   407→   407→              l.done = true;
   408→   408→              l.doneT = 0;
   409→   409→              say('The golden delicious!', { rate: 1.0, pitch: 1.2 }); // show line
   410→   410→              sfxFanfare();
   411→   411→              spawnSparkles(this.particles, l.laneX, l.y - 110, 60, PAL.goldLight);
   412→   412→            }
   413→   413→          } else if (a.y > 900) {
   414→   414→            a.dead = true;

  1224→  1224→
