   136→  // ---------- input ----------
   137→  onTap(x: number, y: number): void {
   138→    initAudio();
   139→    pickVoice();
   140→    // mute button hit test (top-right circle)
   141→    const mx = 522, my = 46, mr = 30;
   142→    if ((x - mx) ** 2 + (y - my) ** 2 <= mr * mr) {
   143→      this.mutedFlag = !isMuted();
   144→      setMuted(this.mutedFlag);
   145→      if (!this.mutedFlag) sfxTap();
   146→      return;
   147→    }
   148→    switch (this.state) {
   149→      case 'title':
   150→        startMusic();
   151→        this.enterIntro(1);
   152→        break;
   153→      case 'intro':
   154→        this.startLevel(this.level);
   155→        break;
   156→      case 'reveal':
   157→        break; // auto-advances
   158→      case 'retry':
   159→        break; // auto-advances
   160→      case 'win':
   161→        this.backToTitle();
   162→        break;
   163→      case 'play':
   164→        this.tapPlay(x, y);
   165→        break;
   166→    }
   167→  }

   169→  private tapPlay(x: number, y: number): void {

   171→    else if (this.level === 2) this.tapL2(x);
   172→    else if (this.level === 3) this.tapL3();
   173→    else if (this.level === 4) this.tapL4();
   174→    else if (this.level === 5) this.tapL5();
   175→    else if (this.level === 6) this.tapL6();
   176→  }

   249→  private tapL1(x: number): void {
   250→    const l = this.l1;
   251→    if (l.merged || l.done) return;
   252→    const side = x < W / 2 ? -1 : 1; // -1 left, 1 right
   253→    // NOTE: positive angle = clockwise = RIGHT side down, LEFT side up
   254→    const highSide = l.angle > 0.035 ? -1 : l.angle < -0.035 ? 1 : (Math.random() > 0.5 ? -1 : 1);
   255→    const good = side === highSide || Math.abs(l.angle) <= 0.035;
   256→    if (good) {
   257→      l.count++;
   258→      l.angVel += side * 1.5;
   259→      if (side < 0) l.bounceL = 1; else l.bounceR = 1;
   260→      sfxTap(1 + (l.count % 3) * 0.12);
   261→      sfxBoing(side < 0 ? 1.1 : 0.9);
   262→      sayNumber(Math.min(8, l.count));
   263→      spawnSparkles(this.particles, W / 2 + side * 180, 620, 6, PAL.goldLight);
   264→      if (l.count === 4) say('Keep going! Keep going!', { rate: 1.15 });
   265→      if (l.count >= 8) {
   266→        l.merged = true;
   267→        l.mergeT = 0;
   268→        say('And double it! Four plus four equals eight!', { rate: 1.05 });
   269→        sfxSparkle();
   270→        spawnSparkles(this.particles, W / 2, 560, 50, PAL.goldLight);
   271→      }
   272→    } else {
   273→      l.angVel += side * 1.9;
   274→      sfxBonk();
   275→      spawnSparkles(this.particles, W / 2 + side * 180, 620, 8, PAL.redLight);
   276→    }
   277→  }
   278→
   279→  private updateL1(dt: number): void {
   280→    const l = this.l1;
   281→    l.windT += dt;
   282→    if (l.bounceL > 0) l.bounceL = Math.max(0, l.bounceL - dt * 4);
   283→    if (l.bounceR > 0) l.bounceR = Math.max(0, l.bounceR - dt * 4);
   284→    if (!l.merged) {
   285→      // destabilizing beam physics + gentle wind gusts (tuned to be fair)
   286→      l.angVel += (l.angle * 1.2 + Math.sin(l.windT * 1.7) * 0.35) * dt * 2.0;
   287→      l.angVel *= 0.955;
   288→      l.angle += l.angVel * dt * 0.5;
   289→      if (Math.abs(l.angle) > 0.72) {
   290→        this.enterRetry('THE BLOCKS SLID OFF!');
   291→        return;
   292→      }
   293→    } else {
   294→      l.mergeT += dt;
   295→      if (l.mergeT > 2.4 && !l.done) {
   296→        l.done = true;
   297→        this.enterReveal(['4', '+', '4', '=', '8'], () => {
   298→          this.enterIntro(2);
   299→        });
   300→      }
   301→    }
   302→  }

   304→  private drawL1(): void {
   305→    const c = this.c;
   306→    const l = this.l1;
   307→    drawCaveBackground(c, this.globalT);
   308→    drawSpotlight(c, W / 2, 0, 260, 780, 0.22);

   311→    // seesaw
   312→    const angle = l.merged ? 0 : l.angle;
   313→    drawSeesaw(c, 288, 640, 216, angle);
   314→    // the two green 4s riding the beam (seated along the beam line)
   315→    const beamY = 640;
   316→    const sa = Math.sin(angle);
   317→    const ca = Math.cos(angle);
   318→    if (!l.merged) {
   319→      const squashL = l.bounceL * 0.5;
   320→      const squashR = l.bounceR * 0.5;
   321→      drawGreen4(c, {
   322→        cx: 288 - 200 * ca, bottomY: beamY - 200 * sa - 9,
   323→        scale: 0.78, look: 0, mouth: Math.abs(l.angle) > 0.4 ? 'worried' : 'smile',
   324→        squash: squashL, tilt: angle, time: this.globalT,
   325→      });
   326→      drawGreen4(c, {
   327→        cx: 288 + 200 * ca, bottomY: beamY + 200 * sa - 9,
   328→        scale: 0.78, look: 0, mouth: Math.abs(l.angle) > 0.4 ? 'worried' : 'smile',
   329→        squash: squashR, tilt: angle, time: this.globalT,
   330→      });
   331→    } else {
   332→      // merge animation: both slide to center, pink 8 rises
   333→      const mt = Math.min(1, l.mergeT / 1.0);
   334→      const gx = 288 - 200 * (1 - mt);
   335→      const gy = beamY - 200 * sa * (1 - mt);
   336→      if (mt < 1) {
   337→        drawGreen4(c, { cx: gx, bottomY: gy, scale: 0.78, look: 0, squash: 0.3, armsUp: true, time: this.globalT, alpha: 1 - mt * 0.7 });
   338→        drawGreen4(c, { cx: 288 + 200 * (1 - mt), bottomY: gy, scale: 0.78, look: 0, squash: 0.3, armsUp: true, time: this.globalT, alpha: 1 - mt * 0.7 });
   339→      }
   340→      if (l.mergeT > 0.9) {
   341→        const rise = Math.min(1, (l.mergeT - 0.9) / 0.7);
   342→        const bob = Math.sin(this.globalT * 3) * 6;
   343→        const bottom = beamY + 20 - rise * 160 + bob * rise;
   344→        drawPink8(c, {
   345→          cx: 288, bottomY: bottom, scale: 0.85 + rise * 0.15,
   346→          look: 0, armsUp: true, apples: 0, mouth: 'smile', time: this.globalT,
   347→        });
   348→        drawBadge(c, 288, bottom - 300, 8, 1.1);
   349→      }
   350→    }
   351→    // equation chips
   352→    const chips: Array<'4' | '+' | '=' | '8' | '?'> = ['4', '+', '4', '=', l.count >= 8 ? '8' : '?'];
   353→    drawChipEquation(c, 92, 96, chips);
   354→    // balance meter
   355→    drawProgressBar(c, 92, 170, 240, 22, l.count / 8);
   356→    this.drawHint('TAP THE HIGH SIDE!');
   357→    this.drawMuteBtn();
   358→    this.drawParticlesAndShake();
   359→  }

   364→  private tapL2(x: number): void {
   365→    const l = this.l2;

   371→  private updateL2(dt: number): void {
   372→    const l = this.l2;
   373→    l.laneX += (l.targetX - l.laneX) * Math.min(1, dt * 6);
   374→    if (!l.done) {
   375→      l.y -= 42 * dt; // auto climb
   376→      if (l.y < 300) l.y = 300;
   377→      l.climbBob += dt;
   378→      // spawn apples
   379→      l.spawnT -= dt;
   380→      if (l.spawnT <= 0 && l.caught + l.apples.filter((a) => !a.caught && !a.dead).length < 9) {
   381→        const lanes = [168, 288, 408];
   382→        l.apples.push({
   383→          x: lanes[Math.floor(Math.random() * 3)] + (Math.random() * 30 - 15),
   384→          y: -40, vy: 150 + Math.random() * 70, caught: false, dead: false, t: 0,
   385→        });
   386→        l.spawnT = Math.max(0.45, 1.1 - this.globalT * 0.01) + Math.random() * 0.5;
   387→      }
   388→      // move & catch
   389→      for (const a of l.apples) {
   390→        a.t += dt;
   391→        if (a.dead) continue;
   392→        if (a.caught) {
   393→          a.x += (l.laneX - a.x) * dt * 8;
   394→          a.y += (l.y - 150 - a.y) * dt * 8;
   395→          if (a.y < l.y - 145) a.dead = true;
   396→        } else {
   397→          a.y += a.vy * dt;
   398→          const dx = a.x - l.laneX;
   399→          const dy = a.y - (l.y - 110);
   400→          if (dx * dx + dy * dy < 80 * 80) {
   401→            a.caught = true;
   402→            l.caught++;
   403→            sfxPop(1 + l.caught * 0.07);
   404→            sayNumber(l.caught);
   405→            spawnSparkles(this.particles, a.x, a.y, 12, PAL.goldLight);
   406→            if (l.caught === 8) {
   407→              l.done = true;
   408→              l.doneT = 0;
   409→              say('The golden delicious!', { rate: 1.0, pitch: 1.2 });
   410→              sfxFanfare();
   411→              spawnSparkles(this.particles, l.laneX, l.y - 110, 60, PAL.goldLight);
   412→            }
   413→          } else if (a.y > 900) {
   414→            a.dead = true;
   415→          }
   416→        }
   417→      }
   418→      l.apples = l.apples.filter((a) => !a.dead);
   419→    } else {
   420→      l.doneT += dt;
   421→      if (l.doneT > 2.3) {
   422→        this.enterReveal(['8', 'GOLDEN', 'APPLES'], () => this.enterIntro(3));
   423→      }
   424→    }
   425→  }

   459→    drawAppleCounter(c, 100, 40, l.caught, 8);
   460→    if (l.done) drawBanner(c, 'THE GOLDEN DELICIOUS!', 940, this.stateT, PAL.goldLight, 36);

   468→  // ============================================================
   469→  private tapL3(): void {
   470→    const l = this.l3;
   471→    if (l.done || l.splitT >= 0) return;
   472→    const onBridge = l.b8x >= l.bridgeStart + 30 && l.b8x <= l.bridgeEnd - 60;
   473→    if (onBridge) {
   474→      l.splitT = 0;
   475→      stopRumble();
   476→      sfxSparkle();
   477→      sfxWhoosh();
   478→      say('Eight equals four plus four!', { rate: 1.05 });

  1267→  private loop(ts: number): void {
  1268→    this.raf = requestAnimationFrame(this.loop);
  1269→    if (!this.lastT) this.lastT = ts;
  1270→    let dt = (ts - this.lastT) / 1000;
  1271→    this.lastT = ts;
  1272→    if (dt > 0.05) dt = 0.05;
  1273→    this.globalT += dt;
  1274→    this.stateT += dt;
  1275→
  1276→    const c = this.c;
  1277→    // scale design space
  1278→    const dpr = Math.min(2, window.devicePixelRatio || 1);
  1279→    c.setTransform(dpr, 0, 0, dpr, 0, 0);
  1280→    const rect = this.canvas.getBoundingClientRect();
  1281→    const sx = rect.width / W;
  1282→    const sy = rect.height / H;
  1283→    c.setTransform(dpr * sx, 0, 0, dpr * sy, 0, 0);
  1284→
  1285→    // letterbox bg
  1286→    c.fillStyle = PAL.bgDark;
  1287→    c.fillRect(0, 0, W, H);
  1288→
  1289→    updateParticles(this.particles, dt * this.timeScale, 300);
  1290→
  1291→    switch (this.state) {
  1292→      case 'title':
  1293→        this.drawTitle();
  1294→        break;
  1295→      case 'intro':
  1296→        this.drawIntro();
  1297→        break;
  1298→      case 'reveal': {
  1299→        this.drawReveal();
  1300→        if (this.stateT > 1.9 && this.pendingNext) {
  1301→          const next = this.pendingNext;
  1302→          this.pendingNext = null;
  1303→          next();
  1304→        }
  1305→        break;
  1306→      }
  1307→      case 'retry': {
  1308→        // keep drawing the level behind
  1309→        this.drawLevelFrozen();
  1310→        this.drawRetry();
  1311→        if (this.stateT > 1.4) {
  1312→          this.startLevel(this.level);
  1313→        }
  1314→        break;
  1315→      }
  1316→      case 'win':
  1317→        this.drawL6();
  1318→        break;
  1319→      case 'play':
  1320→        this.updatePlay(dt);
  1321→        this.drawPlay();
  1322→        break;
  1323→    }
  1324→  }
  1325→
  1326→  private drawLevelFrozen(): void {

  1330→    else if (this.level === 3) this.drawL3();
  1331→    else if (this.level === 4) this.drawL4();
  1332→    else if (this.level === 5) this.drawL5();
  1333→    else this.drawL6();
  1334→  }
