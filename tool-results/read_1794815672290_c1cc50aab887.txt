   111→  constructor(canvas: HTMLCanvasElement) {

   142→  // ---------- input ----------
   143→  onTap(x: number, y: number): void {
   144→    initAudio();
   145→    pickVoice();
   146→    // mute button hit test (top-right circle)
   147→    const mx = 522, my = 46, mr = 30;
   148→    if ((x - mx) ** 2 + (y - my) ** 2 <= mr * mr) {
   149→      this.mutedFlag = !isMuted();
   150→      setMuted(this.mutedFlag);
   151→      if (!this.mutedFlag) sfxTap();
   152→      return;
   153→    }
   154→    switch (this.state) {
   155→      case 'title':
   156→        startMusic();
   157→        this.enterIntro(1);
   158→        break;
   159→      case 'intro':
   160→        this.startLevel(this.level);
   161→        break;
   162→      case 'reveal':
   163→        break; // auto-advances
   164→      case 'retry':
   165→        break; // auto-advances
   166→      case 'win':
   167→        this.backToTitle();
   168→        break;
   169→      case 'play':
   170→        this.tapPlay(x, y);
   171→        break;
   172→    }
   173→  }

   175→  private tapPlay(x: number, y: number): void {
   176→    if (this.level === 1) this.tapL1(x);
   177→    else if (this.level === 2) this.tapL2(x);
   178→    else if (this.level === 3) this.tapL3();
   179→    else if (this.level === 4) this.tapL4();
   180→    else if (this.level === 5) this.tapL5();
   181→    else if (this.level === 6) this.tapL6();
   182→  }

   193→    this.stateT = 0;

   253→  private tapL1(x: number): void {
   254→    const l = this.l1;
   255→    if (l.merged || l.done) return;
   256→    const side = x < W / 2 ? -1 : 1; // -1 left, 1 right
   257→    // NOTE: positive angle = clockwise = RIGHT side down, LEFT side up
   258→    const highSide = l.angle > 0.035 ? -1 : l.angle < -0.035 ? 1 : (Math.random() > 0.5 ? -1 : 1);
   259→    const good = side === highSide || Math.abs(l.angle) <= 0.035;
   260→    if (good) {
   261→      l.count++;
   262→      l.angVel += side * 1.5;
   263→      if (side < 0) l.bounceL = 1; else l.bounceR = 1;
   264→      sfxTap(1 + (l.count % 3) * 0.12);
   265→      sfxBoing(side < 0 ? 1.1 : 0.9);
   266→      sayNumber(Math.min(8, l.count));
   267→      spawnSparkles(this.particles, W / 2 + side * 180, 620, 6, PAL.goldLight);
   268→      if (l.count >= 8) {
   269→        l.merged = true;
   270→        l.mergeT = 0;
   271→        say('And double it! Four plus four equals eight!', { rate: 1.05 });
   272→        sfxSparkle();
   273→        spawnSparkles(this.particles, W / 2, 560, 50, PAL.goldLight);
   274→      }
   275→    } else {
   276→      l.angVel += side * 1.9;
   277→      sfxBonk();
   278→      spawnSparkles(this.particles, W / 2 + side * 180, 620, 8, PAL.redLight);
   279→    }
   280→  }
   281→
   282→  private updateL1(dt: number): void {
   283→    const l = this.l1;
   284→    l.windT += dt;
   285→    if (l.bounceL > 0) l.bounceL = Math.max(0, l.bounceL - dt * 4);
   286→    if (l.bounceR > 0) l.bounceR = Math.max(0, l.bounceR - dt * 4);
   287→    if (!l.merged) {
   288→      // destabilizing beam physics + gentle wind gusts (tuned to be fair)
   289→      l.angVel += (l.angle * 1.2 + Math.sin(l.windT * 1.7) * 0.35) * dt * 2.0;
   290→      l.angVel *= 0.955;
   291→      l.angle += l.angVel * dt * 0.5;
   292→      if (Math.abs(l.angle) > 0.72) {
   293→        this.enterRetry('THE BLOCKS SLID OFF!');
   294→        return;
   295→      }
   296→    } else {
   297→      l.mergeT += dt;
   298→      if (l.mergeT > 2.4 && !l.done) {
   299→        l.done = true;
   300→        this.enterReveal(['4', '+', '4', '=', '8'], () => {
   301→          this.enterIntro(2);
   302→        });
   303→      }
   304→    }
   305→  }

   307→  private drawL1(): void {
   308→    const c = this.c;
   309→    const l = this.l1;
   310→    drawCaveBackground(c, this.globalT);
   311→    drawSpotlight(c, W / 2, 0, 260, 780, 0.22);
   312→    drawGoldenDoor(c, 288, 640, 96, 150, 0.4 + 0.6 * Math.abs(Math.sin(this.globalT * 0.7)));
   313→    drawBallPit(c, 770, 254, this.globalT);
   314→    // seesaw
   315→    const angle = l.merged ? 0 : l.angle;
   316→    drawSeesaw(c, 288, 640, 216, angle);
   317→    // the two green 4s riding the beam (seated along the beam line)
   318→    const beamY = 640;
   319→    const sa = Math.sin(angle);
   320→    const ca = Math.cos(angle);
   321→    if (!l.merged) {
   322→      const squashL = l.bounceL * 0.5;
   323→      const squashR = l.bounceR * 0.5;
   324→      drawGreen4(c, {
   325→        cx: 288 - 200 * ca, bottomY: beamY - 200 * sa - 9,
   326→        scale: 0.78, look: 0, mouth: Math.abs(l.angle) > 0.4 ? 'worried' : 'smile',
   327→        squash: squashL, tilt: angle, time: this.globalT,
   328→      });
   329→      drawGreen4(c, {
   330→        cx: 288 + 200 * ca, bottomY: beamY + 200 * sa - 9,
   331→        scale: 0.78, look: 0, mouth: Math.abs(l.angle) > 0.4 ? 'worried' : 'smile',
   332→        squash: squashR, tilt: angle, time: this.globalT,
   333→      });
   334→    } else {
   335→      // merge animation: both slide to center, pink 8 rises
   336→      const mt = Math.min(1, l.mergeT / 1.0);
   337→      const gx = 288 - 200 * (1 - mt);
   338→      const gy = beamY - 200 * sa * (1 - mt);
   339→      if (mt < 1) {
   340→        drawGreen4(c, { cx: gx, bottomY: gy, scale: 0.78, look: 0, squash: 0.3, armsUp: true, time: this.globalT, alpha: 1 - mt * 0.7 });
   341→        drawGreen4(c, { cx: 288 + 200 * (1 - mt), bottomY: gy, scale: 0.78, look: 0, squash: 0.3, armsUp: true, time: this.globalT, alpha: 1 - mt * 0.7 });
   342→      }
   343→      if (l.mergeT > 0.9) {
   344→        const rise = Math.min(1, (l.mergeT - 0.9) / 0.7);
   345→        const bob = Math.sin(this.globalT * 3) * 6;
   346→        const bottom = beamY + 20 - rise * 160 + bob * rise;
   347→        drawPink8(c, {
   348→          cx: 288, bottomY: bottom, scale: 0.85 + rise * 0.15,
   349→          look: 0, armsUp: true, apples: 0, mouth: 'smile', time: this.globalT,
   350→        });
   351→        drawBadge(c, 288, bottom - 300, 8, 1.1);
   352→      }
   353→    }
   354→    // equation, exactly like the video: "4 + 4"
   355→    const chips: Array<'4' | '+' | '=' | '8' | '?'> = l.merged ? ['4', '+', '4', '=', '8'] : ['4', '+', '4'];
   356→    drawChipEquation(c, 92, 96, chips);
   357→    this.drawMuteBtn();
   358→    this.drawParticlesAndShake();
   359→  }

   364→  private tapL2(x: number): void {
   365→    const l = this.l2;
   366→    if (l.done) return;
   367→    l.targetX = x < 192 ? 168 : x < 384 ? 288 : 408;

   371→  private updateL2(dt: number): void {
   372→    const l = this.l2;
   373→    l.laneX += (l.targetX - l.laneX) * Math.min(1, dt * 6);
   374→    if (!l.done) {
   375→      l.y -= 42 * dt; // auto climb
   376→      if (l.y < 300) l.y = 300;
   377→      l.climbBob += dt;
   378→      // spawn apples
   379→      l.spawnT -= dt;
   380→      if (l.spawnT <= 0 && l.caught + l.apples.filter((a) => !a.caught && !a.dead).length < 9) {
   381→        const lanes = [168, 288, 408];
   382→        l.apples.push({
   383→          x: lanes[Math.floor(Math.random() * 3)] + (Math.random() * 30 - 15),
   384→          y: -40, vy: 150 + Math.random() * 70, caught: false, dead: false, t: 0,
   385→        });
   386→        l.spawnT = Math.max(0.45, 1.1 - this.globalT * 0.01) + Math.random() * 0.5;
   387→      }
   388→      // move & catch
   389→      for (const a of l.apples) {
   390→        a.t += dt;
   391→        if (a.dead) continue;
   392→        if (a.caught) {
   393→          a.x += (l.laneX - a.x) * dt * 8;
   394→          a.y += (l.y - 150 - a.y) * dt * 8;
   395→          if (a.y < l.y - 145) a.dead = true;
   396→        } else {
   397→          a.y += a.vy * dt;
   398→          const dx = a.x - l.laneX;
   399→          const dy = a.y - (l.y - 110);
   400→          if (dx * dx + dy * dy < 80 * 80) {
   401→            a.caught = true;
   402→            l.caught++;
   403→            sfxPop(1 + l.caught * 0.07);
   404→            sayNumber(l.caught);
   405→            spawnSparkles(this.particles, a.x, a.y, 12, PAL.goldLight);
   406→            if (l.caught === 8) {
   407→              l.done = true;
   408→              l.doneT = 0;
   409→              say('The golden delicious!', { rate: 1.0, pitch: 1.2 }); // show line
   410→              sfxFanfare();
   411→              spawnSparkles(this.particles, l.laneX, l.y - 110, 60, PAL.goldLight);
   412→            }
   413→          } else if (a.y > 900) {
   414→            a.dead = true;
   415→          }
   416→        }
   417→      }
   418→      l.apples = l.apples.filter((a) => !a.dead);
   419→    } else {
   420→      l.doneT += dt;
   421→      if (l.doneT > 2.3) {
   422→        this.enterReveal(['8'], () => this.enterIntro(3));
   423→      }
   424→    }
   425→  }

   465→  // ============================================================
   466→  private tapL3(): void {
   467→    const l = this.l3;
   468→    if (l.done || l.splitT >= 0) return;
   469→    const onBridge = l.b8x >= l.bridgeStart + 30 && l.b8x <= l.bridgeEnd - 60;
   470→    if (onBridge) {
   471→      l.splitT = 0;
   472→      stopRumble();
   473→      sfxSparkle();
   474→      sfxWhoosh();
   475→      say('Eight equals four plus four!', { rate: 1.05 });
   476→    } else {
   477→      sfxBonk();
   478→      this.showTooEarlyFlash = 0.8;
   479→    }
   480→  }
   481→  private showTooEarlyFlash = 0;
   482→
   483→  private updateL3(dt: number): void {
   484→    const l = this.l3;
   485→    this.showTooEarlyFlash = Math.max(0, this.showTooEarlyFlash - dt);
   486→    if (l.done) {
   487→      l.walkT += dt;
   488→      if (l.walkT > 2.6) {
   489→        this.enterReveal(['8', '=', '4', '+', '4'], () => this.enterIntro(4));
   490→      }
   491→      return;
   492→    }
   493→    if (l.splitT >= 0) {

  1207→  private loop(ts: number): void {
  1208→    this.raf = requestAnimationFrame(this.loop);
  1209→    if (!this.lastT) this.lastT = ts;
  1210→    let dt = (ts - this.lastT) / 1000;
  1211→    this.lastT = ts;
  1212→    if (dt > 0.05) dt = 0.05;
  1213→    this.globalT += dt;
  1214→    this.stateT += dt;
  1215→
  1216→    const c = this.c;
  1217→    // scale design space
  1218→    const dpr = Math.min(2, window.devicePixelRatio || 1);
  1219→    c.setTransform(dpr, 0, 0, dpr, 0, 0);
  1220→    const rect = this.canvas.getBoundingClientRect();
  1221→    const sx = rect.width / W;
  1222→    const sy = rect.height / H;
  1223→    c.setTransform(dpr * sx, 0, 0, dpr * sy, 0, 0);
  1224→
  1225→    // letterbox bg
  1226→    c.fillStyle = PAL.bgDark;
  1227→    c.fillRect(0, 0, W, H);
  1228→
  1229→    updateParticles(this.particles, dt * this.timeScale, 300);
  1230→
  1231→    switch (this.state) {
  1232→      case 'title':
  1233→        this.drawTitle();
  1234→        break;
  1235→      case 'intro':
  1236→        this.drawIntro();
  1237→        if (this.stateT > 1.6) this.startLevel(this.level); // auto-advance, no words
  1238→        break;
  1239→      case 'reveal': {
  1240→        this.drawReveal();
  1241→        if (this.stateT > 1.9 && this.pendingNext) {
  1242→          const next = this.pendingNext;
  1243→          this.pendingNext = null;
  1244→          next();
  1245→        }
  1246→        break;
  1247→      }
  1248→      case 'retry': {
  1249→        // keep drawing the level behind
  1250→        this.drawLevelFrozen();
  1251→        this.drawRetry();
  1252→        if (this.stateT > 1.4) {
  1253→          this.startLevel(this.level);
  1254→        }
  1255→        break;
  1256→      }
  1257→      case 'win':
  1258→        this.drawL6();
  1259→        break;
  1260→      case 'play':
  1261→        this.updatePlay(dt);
  1262→        this.drawPlay();
  1263→        break;
  1264→    }
  1265→  }
  1266→
  1267→  private drawLevelFrozen(): void {
  1268→    // draw the last level frame statically behind retry overlay
  1269→    if (this.level === 1) this.drawL1();
  1270→    else if (this.level === 2) this.drawL2();
  1271→    else if (this.level === 3) this.drawL3();
  1272→    else if (this.level === 4) this.drawL4();
  1273→    else if (this.level === 5) this.drawL5();
  1274→    else this.drawL6();
  1275→  }
  1276→
  1277→  private updatePlay(dt: number): void {
  1278→    const d = dt * this.timeScale;
  1279→    if (this.level === 1) this.updateL1(d);
  1280→    else if (this.level === 2) this.updateL2(d);
  1281→    else if (this.level === 3) this.updateL3(d);
  1282→    else if (this.level === 4) this.updateL4(d);
  1283→    else if (this.level === 5) this.updateL5(d);
  1284→    else if (this.level === 6) this.updateL6(d);
  1285→  }
  1286→
  1287→  private drawPlay(): void {
  1288→    if (this.level === 1) this.drawL1();
  1289→    else if (this.level === 2) this.drawL2();
  1290→    else if (this.level === 3) this.drawL3();
  1291→    else if (this.level === 4) this.drawL4();
  1292→    else if (this.level === 5) this.drawL5();
  1293→    else if (this.level === 6) this.drawL6();
  1294→  }
